    21	export function parseCsv(texto: string): ParsedCsv {
    22	  const linhas = texto.split(/\r?\n/).filter((l) => l.trim().length > 0);
    23	  if (linhas.length === 0) return { cabecalho: [], linhas: [] };
    24
    25	  const delimitador = linhas[0].includes(';') ? ';' : ',';
    26
    27	  const parseLinha = (linha: string): string[] => {
    28	    const cols: string[] = [];
    29	    let atual = '';
    30	    let entreAspas = false;
    31	    for (let i = 0; i < linha.length; i += 1) {
    32	      const ch = linha[i];
    33	      if (ch === '"') {
    34	        entreAspas = !entreAspas;
    35	      } else if (ch === delimitador && !entreAspas) {
    36	        cols.push(atual.trim());
    37	        atual = '';
    38	      } else {
    39	        atual += ch;
    40	      }
    41	    }
    42	    cols.push(atual.trim());
    43	    return cols.map((c) => c.replace(/^"|"$/g, '').trim());
    44	  };
    45
    46	  const cabecalho = parseLinha(linhas[0]);
    47	  const dados = linhas
    48	    .slice(1)
    49	    .map(parseLinha)
    50	    .filter((l) => l.some((c) => c !== ''));
    51
    52	  return { cabecalho, linhas: dados };
